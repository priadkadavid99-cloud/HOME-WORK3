import tkinter as tk
from tkinter import messagebox


win = tk.Tk()
win.geometry("500x550")

win.config(bg="#199c79")

score, time_left, playing = 0, 10, False
bomb_img = tk.PhotoImage(file="bomb.png.png")
boom_img = tk.PhotoImage(file="pow.png.png")

time_label = tk.Label(
    win, text="ЧАС: 10", font=("Arial", 14, "bold"), bg="#199c79"
)
score_label = tk.Label(
    win, text="БАЛИ: 0", font=("Arial", 14, "bold"), bg="#199c79"
)
pic_label = tk.Label(win, image=bomb_img, bg="#199c79")


def click(event):
  global score
  if playing:
    score += 1
    score_label.config(text=f"БАЛИ: {score}")


def tick():
  global time_left, playing
  if not playing:
    return
  time_label.config(text=f"ЧАС: {time_left}")
  if time_left:
    time_left -= 1
    win.after(1000, tick)
  else:
    playing = False
    pic_label.config(image=boom_img)


def start():
  global score, time_left, playing
  messagebox.showinfo("Правила", "Тисни на бомбу! 10 секунд.")
  start_btn.pack_forget()
  score, time_left, playing = 0, 10, True
  time_label.place(x=20, y=15)
  score_label.place(x=290, y=15)
  pic_label.pack(expand=True)
  tick()


pic_label.bind("<Button-1>", click)
start_btn = tk.Button(
    win, text="ПОЧАТИ ГРУ", font=("Arial", 16, "bold"), command=start
)
start_btn.pack(expand=True)
win.mainloop()
